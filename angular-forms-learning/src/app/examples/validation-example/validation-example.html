カスタムバリデータのコード例:

function forbiddenWordValidator(words: string[]): ValidatorFn {
  return (control: AbstractControl): ValidationErrors | null => {
    const hasForbidden = words.some(w =>
      control.value?.includes(w)
    );
    return hasForbidden ? { forbiddenWord: true } : null;
  };
}

// 使用例
this.fb.group({
  username: ['', forbiddenWordValidator(['admin', 'root'])]
});
